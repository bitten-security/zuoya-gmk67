name: Release

on:
  push:
    tags: ["v*"]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  windows:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install -r requirements.txt pyinstaller
      - run: powershell -File build_exe.ps1
      - name: Installer
        shell: pwsh
        run: |
          choco install innosetup -y --no-progress
          $v = "${{ github.ref_name }}".TrimStart("v")
          if (-not $v -or $v -eq "main") { $v = "0.0.0" }
          & "${env:ProgramFiles(x86)}\Inno Setup 6\ISCC.exe" "/DAppVersion=$v" installer.iss
      - uses: softprops/action-gh-release@v2
        if: startsWith(github.ref, 'refs/tags/v')
        with:
          files: |
            installer/GMK67Control-Setup.exe
            dist/GMK67Control.exe
